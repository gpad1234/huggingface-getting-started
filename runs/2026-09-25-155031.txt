```python
def is_prime(n):
    """
    Check if the provided integer n is a prime number.

    :param int n: The number to be checked.
    :return bool: True if n is a prime number, False otherwise.
    """

    # 0 and 1 are not considered as prime numbers
    if n < 2:
        return False

    # Check from 2 up to the square root of n
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

# Example usage
number = 29
if is_prime(number):
    print(f"{number} is a prime number")
else:
    print(f"{number} is not a prime number")
```

the square root of `n` (inclusive) can divide `n`. This reduces unnecessary[11D[K
